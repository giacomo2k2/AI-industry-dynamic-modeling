\section{Conclusions and Future Works}

\begin{frame}{Conclusions and Future Works}
    \begin{itemize}
        \item \textbf{Chaotic Challenge}: It is difficult to predict a chaotic model after several steps.
        \item \textbf{Metrics}: It is better to focus on metrics other than state prediction, as \textbf{parameter Identification} and \textbf{power spectral density} for instance.
    \end{itemize}
    \begin{block}{Other points}
        \begin{itemize}
            \item \textbf{Anomaly Detection} of degraded models or physically wrong ones.
            \item Developement of a \textbf{merged pipeline} that puts together identification and energy conservation.
        \end{itemize}
    \end{block}
\end{frame}
