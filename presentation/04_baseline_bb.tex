\begin{frame}{Black Box Approaches - Architecture}
    Standard Neural Network for setting a \textbf{baseline}. The objective is to learn the \textbf{dynamics} of the double pendulum. 

    \begin{figure}[h]
        \centering
        \includegraphics[width=0.6\linewidth]{04_baseline_bb/DynNN.pdf} \\
        \vspace{0.3em}
    \end{figure}    %

    The model takes as inputs the \textbf{states} and outputs the \textbf{derivatives}. 

    \begin{block}{Training methods}

        \begin{itemize}
            \item \textbf{Derivatives}: loss is computed directly on the derivatives.
            \item \textbf{Evolution}: the derivatives are fed to a \textbf{solver} that outputs the next state. A sequential stream of predicted
            states is used to compute the loss. 
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Black Box Approaches - Training recipe (1)}
    

    \begin{block}{Trigonometric Transformation}
        Since $\theta \in \bigl[-\pi, \pi\bigr]$, it's impossible to use the angles directly as inputs without introducing \textbf{discontinuities}.
        The solution is to transform the input space.

        \[
            \bigl[\theta_1, \theta_2, \dot{\theta}_1, \dot{\theta}_2\bigr]^{\top}
            \rightarrow
            \bigl[\cos{\theta_1}, \sin{\theta_1}, \cos{\theta_2}, \sin{\theta_2}, \dot{\theta}_1, \dot{\theta}_2\bigr]^{\top}
        \]
    \end{block}

    % TODO: update velocities and acceleration fitted with gaussian and laplacian respectively
    \begin{block}{Normalization}
        \begin{columns}[T]
            \begin{column}{0.33\textwidth}
                \centering
                \vspace{0.5em}
                \textbf{Angles}
                
                \vspace{-1em}

                $$-1 \le \cos{\theta} \le 1$$
                $$-1 \le \sin{\theta} \le 1$$

            \end{column}
            
            \begin{column}{0.33\textwidth}
                \centering
                \vspace{0.5em}
                \textbf{Velocities}

                \begin{figure}[h]
                    \centering
                    \includegraphics[width=0.8\linewidth]{Images/04_baseline_bb/Gaussian.png} \\
                    \vspace{0.3em}
                \end{figure}    

                \vspace{-3em}
        
                $$\hat{\dot{\theta}} = \frac{\dot{\theta} - \mu}{\sigma}$$
            \end{column}
            
            \begin{column}{0.33\textwidth}
                \centering
                \vspace{0.5em}
                \textbf{Accelerations}

                \begin{figure}[h]
                    \centering
                    \includegraphics[width=0.8\linewidth]{Images/04_baseline_bb/Laplacian.png} \\
                    \vspace{0.3em}
                \end{figure}    
                
                \vspace{-3em}
            
                $$\hat{\ddot{\theta}} = \frac{\ddot{\theta} - \tilde{\mu}}{\sqrt{2b}}$$\footnote{with $b = E\bigl[|q - \tilde{\mu}|\bigr]$}
                
                
            \end{column}
        \end{columns}
        
    \end{block}

    
\end{frame}




\begin{frame}{Black Box Approaches - Training recipe (2)}
    
    % \begin{block} {Learning rate}
    %     A learning rate scheduler is used 

    % \end{bloc

    \begin{columns}[T]
            \begin{column}{0.5\textwidth}
              \begin{block} {Sample weights}
                    Computed using the 4D KDE of the input distribution. In particular: 

                    % TODO: update max or total
                    $$\sigma(\theta_1, \theta_2, \dot{\theta}_1, \dot{\theta}_2) = \frac{h(\theta_1, \theta_2, \dot{\theta}_1, \dot{\theta}_2)}{h_{max}}$$

                    It's important to use the multivariate distribution to capture any \textbf{coupling term} in the dynamics

                \end{block}

            \end{column}
            
            \begin{column}{0.3\textwidth}
            \begin{figure}[h]
                    \centering
                    \includegraphics[width=0.6\linewidth]{stock.png} \\
            \end{figure}   

                
            \end{column}
        \end{columns}


    

    \begin{block} {Curriculum learning}
        Used specifically for \textbf{evolution} learning. 

        \par The \textbf{window size} of training is increased periodically during training, usually by doubling. 

        Starting with a window size large from the start is infeasible  due to high gradients. 


        
    \end{block}

\end{frame}
