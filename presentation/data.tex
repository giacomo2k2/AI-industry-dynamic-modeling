\subsection{Preprocessing}

\begin{frame}
    \frametitle{Data Preprocessing Pipeline}
    
    \begin{itemize}
        \item \textbf{Energy Conservation Check:} 
        discard entire simulations where energy deviation exceeds tolerance ($\Delta E > 10^{-2}$).
    \end{itemize}

    \vspace{1em}

    \begin{itemize}
        \item \textbf{Dynamic Time Cut:} 
        truncate simulation duration $T$ to maximize uniformity of state distribution (via KL Divergence).
        \[ T^* = \arg\min_T D_{KL}(P_{\text{data}} || U_{\text{uniform}}) \]
    \end{itemize}

    \vspace{1em}

    \begin{itemize}
        \item \textbf{Noise Injection:} add Gaussian noise $\mathcal{N}(0, 0.01)$ to states.
        \item \textbf{Angle Wrapping:} map angles to $[-\pi, \pi]$ via modulo operation.
    \end{itemize}

\end{frame}
