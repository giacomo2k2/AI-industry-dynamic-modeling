\begin{frame}{Idea of the project}
    \begin{itemize}
        \item \textbf{Problem context}: modeling non-linear dynamical systems
        \item \textbf{Why a double pendulum}: well-known, relatively simple dynamics, chaotic behavior
        \item \textbf{Industrial relevance}:
        \begin{itemize}
            \item Benchmark for ML \& DL for physical systems
            \item Applicable to more complex fields (e.g. \textbf{robotics}, \textbf{aerospace})
        \end{itemize}
        \item \textbf{Applications}: control of multi-articulated systems, energy optimization, short-term prediction
    \end{itemize}
