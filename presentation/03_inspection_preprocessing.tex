\section{Data Inspection}

\begin{frame}{Data Inspection}
    \textbf{Why is Data Inspection crucial?}

    Before training any model, we must validate the physical consistency and statistical quality of our simulation data
    
    \vspace{1em}

    \textbf{Our Inspection Pipeline:}
    \begin{enumerate}
        \item \textbf{Physics Validation:} ensuring energy conservation laws hold (for frictionless cases)
        \item \textbf{Distribution Analysis:} checking for class imbalance or skewness in state variables
        \item \textbf{Stationarity Check:} verifying that statistical properties (mean, variance) remain stable over time
        \item \textbf{Correlation Study:} identifying redundant features to simplify the learning task
    \end{enumerate}

\end{frame}

\begin{frame}{Pre-processing}
    Based on the data inspection results, a preprocessing pipeline was designed to enhance data quality and model training effectiveness
    \begin{itemize}
        \item \textbf{Filtering out evolution with increasing energy}: removing unstable or numerically faulty simulations
        \item \textbf{White Noise}: added noise to simulate reality and add robustness to the system
        \item \textbf{KL Divergence + Cut}:
        \begin{itemize}
            \item Calculation of the \textbf{Kullback-Leibler Divergence} between actual distribution and uniform distribution. Cut of the dataset at the optimum threshold $K^*$ that minimizes the divergence
            \item Identification and removal of the part of the dataset in which the dynamics is less representative
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Data Preprocessing: Dynamic Time Cut Inspection}
    Following the KL Divergence analysis, the dataset was trimmed to exclude less representative dynamics, enhancing model training quality
   
    \begin{block}{Kullback-Leibler Divergence Definition}
        The Kullback-Leibler Divergence between two probability distributions $P$ and $Q$ is defined as:
        \begin{equation}
            D_{KL}(P || Q) = \sum_{i} P(i) \log\left(\frac{P(i)}{Q(i)}\right)
        \end{equation}

        Where: 

        \begin{itemize}
            \item $ Q $ observed distribution
            \item $ P \sim \mathcal{U}$ uniform distribution
        \end{itemize}
        
    \end{block}
\end{frame}



\begin{frame}{Adaptive Dataset Truncation (1)}
    \textbf{Problem:} In high-friction scenarios, the system eventually settles into a static equilibrium. Including these static frames would bias the model towards zero-velocity predictions.
    
    \vspace{0.2cm}
    \textbf{Solution:} We treat truncation as an optimization problem to find the optimal cutoff time $T^*$.
    
    \begin{block}{The Objective Function}
        We minimize a custom loss function combining \textbf{distribution quality} and \textbf{data retention}:
        \centering
        \begin{equation}
            \mathcal{L}(T) = \underbrace{D_{KL}( P || Q_T)}_{\text{Chaos Metric}} + \lambda \cdot \underbrace{\left( \frac{T_{max} - T}{T_{max}} \right)^2}_{\text{Data Loss Penalty}}
        \end{equation}  
        \begin{equation}
            T^* = \arg\min_{T} \mathcal{L}(T)
        \end{equation} 
    \end{block}
\end{frame}

\begin{frame}{Adaptive Dataset Truncation (2)}
    Where:
    \begin{itemize}
        \item $ Q_T $: observed distribution up to time $ T $
        \item $ T_{max} $: total simulation time
        \item $ \lambda $: regularization parameter balancing the two terms
        \item $ T $: candidate truncation time
    \end{itemize}
    
    \vspace{0.2cm}
    \small{\textbf{Result:} The algorithm automatically identifies the exact moment the system loses its ergodic properties.}
    
    \begin{block}{Recall: Ergodicity Hypothesis}
    A dynamical system is \textbf{Ergodic} if it visits every point in the phase space with equal probability over a long period.
    
    \vspace{0.1cm}
    $$ \text{Time Average} \equiv \text{Space Average} $$
    
    \vspace{0.1cm}
    \footnotesize{\textbf{In our context:} Ideally, a frictionless pendulum explores all possible angles and velocities allowed by its energy, resulting in a \textbf{Uniform Distribution} ($P$).}
\end{block}

\end{frame}
