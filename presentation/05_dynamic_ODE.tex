    \begin{block}{Physical Parameters Estimation}
        \begin{itemize}
            \setlength\itemsep{1em} % Adds space between items for readability

            physical parameters are fixed over time (time-invariance)
            
            \item \textbf{Knowledge:} 
            \begin{itemize}
                \item Analytical dynamics 
                \item System state trajectories $(q, \dot{q})$
            \end{itemize}
            
            \item \textbf{Goal:} 
            estimate the unknown parameters
            $$ m_1, m_2, l_1, l_2, b_1, b_2 $$
        \end{itemize}
